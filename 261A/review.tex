\subsection{Starting Out} \label{subsec:start}
\begin{definition}[Lie algebra]
	A \textit{Lie algebra} is a vector space $\mf g$ equipped with a \textit{Lie bracket} $[-,-]\colon\mf g\times\mf g\to\mf g$ satisfying the following properties.
	\begin{listroman}
		\item Bilinear: the map $[-,-]$ is bilinear.
		\item Alternating: $[X,X]=0$ for all $X\in\mf g$.
		\item Jacobi identity: $[X,[Y,Z]]+[Y,[Z,X]]+[Z,[X,Y]]=0$ for all $X,Y,Z\in\mf g$.
	\end{listroman}
	A morphism of Lie algebras is a linear map preserving the bracket.
\end{definition}
\begin{example}
	For any (possibly non-commutative, non-unital) algebra $A$ over $\CC$, one can define the Lie bracket $[-,-]\colon A\times A\to A$ by
	\[[a,b]\coloneqq ab-ba.\]
	For example, applying this to $A=\op{End}(V)$ where $V$ is a vector space gives the Lie algebra $\mf{gl}(V)$. If further $V=\CC^n$, we abbreviate $\mf{gl}(V)$ to $\mf{gl}_n(\CC)$.
\end{example}
\begin{example}
	The subspace
	\[\mf{sl}_n(\CC)\coloneqq\{X\in\mf{gl}_n(\CC):\tr X=0\}\]
	is a Lie subalgebra.
\end{example}
\begin{example}
	The subspace of alternating matrices is a Lie subalgebra, called $\mf o_n(\CC)\subseteq\mf{gl}_n(\CC)$.
\end{example}
\begin{definition}[representation]
	A \textit{representation} of a Lie algebra $\mf g$ is a vector space $V$ along with a morphism $\rho\colon\mf g\to\mf{gl}(V)$. A morphism of representations is a morphism of the underlying vector spaces which preserves the $\mf g$-action.
\end{definition}
\begin{example}[adjoint]
	The adjoint representation of a Lie algebra $\mf g$ is the morphism $\op{ad}_\bullet\colon\mf g\to\mf{gl}(\mf g)$ defined by
	\[\op{ad}_X(Y)\coloneqq[X,Y].\]
	The fact that $\op{ad}_\bullet$ preserves the Lie bracket is equivalent to the Jacobi identity.
\end{example}
\begin{example}
	For any representation $V$ of $\mf g$, there is a dual representation $V^\lor$ whose underlying vector space is the dual, and $X\in\mf g$ acts on $\ell\colon V\to\CC$ by $(X\ell)(v)\coloneqq-\ell(Xv)$.
\end{example}
\begin{example}
	One can take sums of representaions. For two representations $V$ and $W$, one can also take a tensor product: $X\in\mf g$ acts by $X(v\otimes w)=Xv\otimes w+v\otimes Xw$.
\end{example}
\begin{definition}[irreducible]
	A representation $\rho\colon\mf g\to\mf{gl}(V)$ is \textit{irreducible} if and only if $V\ne0$, and $V$ admits no nonzero proper subrepresentations. A representation is \textit{completely reducible} if and only if it is a direct sum of irreducible representations.
\end{definition}
\begin{example}
	Let $\mf b\subseteq\mf{gl}_2(\CC)$ be the Lie subalgebra of upper-triangular matrices. Then the representation $\mf b\to\mf{gl}_2(\CC)$ is not completely reducible.
\end{example}
\begin{lemma}[Schur] \label{lem:schur}
	Let $\rho_V\colon\mf g\to\mf{gl}(V)$ and $\rho_W\colon\mf g\to\mf{gl}(W)$ be representations, and let $\varphi\colon V\to W$ be a nonzero morphism of representations.
	\begin{listalph}
		\item If $V$ is irreducible, then $\varphi$ is injective.
		\item If $W$ is irreducible, then $\varphi$ is surjective.
		\item If both $V$ and $W$ are irreducible, then $\varphi$ is an isomorphism.
	\end{listalph}
\end{lemma}
\begin{proof}[Sketch]
	Here, (c) follows from (a) and (b). We only sketch (a): because $\varphi$ is nonzero, $\ker\varphi$ is a proper subrepresentation of $V$, so $\ker\varphi=0$ because $V$ is irreducible.
\end{proof}
\begin{example}
	Let the natural action of $\mf{sl}_2(\CC)$ on $\CC^2$ is irreducible. It turns out that the irreducible representations of $\mf{sl}_2(\CC)$ are exactly the symmetric powers of $\CC^2$. To see this, show that $\op{diag}(+1,-1)$ acts semisimply on any irreducible representation, and then one can parameterize the irreducible representations by the highest eigenvalue of $\op{diag}(+1,-1)$.
\end{example}
\begin{definition}[universal enveloping algebra]
	Fix a Lie algebra $\mf g$. Then the \textit{universal enveloping algebra} $U\mf g$ of $\mf g$ is the free complex algebra generated by $\mf g$ along with the relations
	\[XY-YX=[X,Y]\]
	for each $X,Y\in\mf g$.
\end{definition}
\begin{remark}
	The data of a representation $\mf g\to\mathfrak{gl}(V)$ is equivalent to the data of a ring homomorphism $U\mf g\to\op{End}(V)$.
\end{remark}
\begin{theorem}[Poincar\'e--Birkhoff--Witt] \label{thm:pbw}
	Fix a Lie algebra $\mf g$, and let $\{X_i\}_{i=1}^n$ be an (ordered) basis. Then the monomials
	\[\{X_{i_1}\cdots X_{i_m}:i_1\le i_2\le\cdots\le i_m\}\]
	form a basis of $U\mf g$.
\end{theorem}
\begin{proof}[Sketch]
	Certainly $U\mf g$ is spanned by monomials, and one can use the relations $XY=YX+[X,Y]$ in order to rearrange the entries of the monomial. Thus, the given monomials span. Checking that they are linearly independent amounts to a long and technical argument.
\end{proof}

\subsection{Nilpotent and Solvable Lie Algebras} \label{subsec:solvable}
\begin{definition}[ideal]
	Fix a Lie algebra $\mf g$. Then an \textit{ideal} $I\subseteq\mf g$ is a subspace such that $[X,I]\subseteq I$ for any $X\in\mf g$. Equivalently, $I$ is a $\mf g$-subrepresentation of $\mf g$.
\end{definition}
\begin{example}[center]
	Fix a Lie algebra $\mf g$. Then the \textit{center} is the subspace
	\[\mf z(\mf g)\coloneqq\{X\in\mf g:[X,\mf g]=0\}.\]
	The center is an ideal. It is kernel the adjoint representation $\op{ad}\colon\mf g\to\mf{gl}(\mf g)$.
\end{example}
\begin{example}[commutator]
	Fix a Lie algebra $\mf g$. Then the \textit{commutator} is the subspace $[\mf g,\mf g]$. By the Jacobi identity, $[\mf g,\mf g]$ is a Lie ideal.
\end{example}
\begin{remark}
	If $I$ is an ideal of a Lie algebra $\mf g$, the quotient space $\mf g/I$ is a Lie algebra, and there is a short exact sequence
	\[0\to I\to\mf g\to\mf g/I\to0\]
	of $\mf g$-modules.
\end{remark}
\begin{definition}
	A Lie algebra $\mf g$ is \textit{abelian} if and only if $[\mf g,\mf g]=0$.
\end{definition}
\begin{example}
	For any Lie algebra, the quotient $\mf g/[\mf g,\mf g]$ is abelian.
\end{example}
\begin{definition}[nilpotent]
	A Lie algebra $\mf g$ is \textit{nilpotent} if and only if there is a sequence
	\[\mf g=\mf a_0\supseteq\mf a_1\supseteq\cdots\supseteq\mf a_k=0\]
	such that $\mf a_{i+1}$ is an ideal of $\mf a_i$ with abelian quotient.
\end{definition}
\begin{remark}
	It is equivalent to check if the sequence $\mf a_{i+1}\coloneqq[\mf g,\mf a_i]$ eventually stabilizes at $0$.
\end{remark}
\begin{example}
	The subspace of strictly upper-triangular matrices of $\mf{gl}_n(\CC)$ is nilpotent.
\end{example}
\begin{definition}[solvable]
	A Lie algebra $\mf g$ is \textit{solvable} if and only if there is a sequence
	\[\mf g=\mf a_0\supseteq\mf a_1\supseteq\cdots\supseteq\mf a_k=0\]
	such that each $\mf a_i$ is an ideal of $\mf g$ such that $\mf g/\mf a_i$ is abelian.
\end{definition}
\begin{remark}
	It is equivalent to check if the sequence $\mf a_i\coloneqq[\mf a_{i+1},\mf a_{i+1}]$ eventually stabilizes at $0$.
\end{remark}
\begin{example}
	Any nilpotent Lie algebra is solvable.
\end{example}
\begin{example}
	The subspace of upper-triangular matrices of $\mf{gl}_n(\CC)$ is solvable but not nilpotent.
\end{example}
\begin{theorem}[Lie] \label{thm:lie}
	Let $\mf g$ be a solvable Lie algebra. For any representation $V$ of $\mf g$, there is a basis in which $\mf g$ acts by upper-triangular matrices.
\end{theorem}
\begin{proof}[Sketch]
	By induction on $\dim V$, it is enough to show that the action of $\mf g$ on $V$ admits a nonzero eigenvector. For this, we induct on $\dim\mf g$. By solvability, one may find a codimension-$1$ ideal $\mf g'\subseteq\mf g$; choose any $X\in\mf g\setminus\mf g'$. Now, induction tells us that $V$ admits a nonzero eigenvector $v$ for $\mf g'$. With some difficulty, can show that
	\[W\coloneqq\op{span}_\CC X^\NN v\]
	is a $\mf g$-subrepresentation of $V$, and it turns out that $\mf g'$ acts diagonally on all of $W$. Thus, any eigenvector of $X$ acting on $W$ will do.
\end{proof}
\begin{corollary}
	Let $\mf g$ be a Lie algebra. Then $\mf g$ is solvable if and only if $[\mf g,\mf g]$ is nilpotent.
\end{corollary}
\begin{proof}[Sketch]
	If $[\mf g,\mf g]$ is nilpotent, then one can show that $\mf g$ is solvable by building ideals from the short exact sequence
	\[0\to[\mf g,\mf g]\to\mf g\to\mf g/[\mf g,\mf g]\to0.\]
	Conversely, if $\mf g$ is solvable, then by \Cref{thm:lie}, one can find a basis of $\mf g$ on which the adjoint action of $\mf g$ on $\mf g$ is given by upper-triangular matrices. Thus, the action of $[\mf g,\mf g]$ is given by strictly upper-triangular matrices. This is enough to show $[\mf g,\mf g]$ is nilpotent.
\end{proof}
\begin{remark} \label{rem:nilp-to-strictly-upper-triangular}
	Similarly, if $\mf g$ is nilpotent, one can show that any representation $V$ admits a basis on which $\mf g$ acts by strictly upper-triangular matrices.
\end{remark}
\begin{theorem}[Engel] \label{thm:engel}
	Let $\mf g$ be a Lie algebra. Then $\mf g$ is nilpotent if and only if the operator $\op{ad}_X\colon\mf g\to\mf g$ is nilpotent for all $X\in\mf g$.
\end{theorem}
\begin{proof}[Sketch]
	If $\op{ad}_X$ is nilpotent for all $X\in\mf g$, then $[\mf g,[\mf g,[\mf g,\ldots,[\mf g,\mf g]]]]=0$ for a high enough nesting level, so $\mf g$ is nilpotent. Conversely, if $\mf g$ is nilpotent, then \Cref{rem:nilp-to-strictly-upper-triangular} grants a basis of $\mf g$ upon which $\mf g$ acts by strictly upper-triangular matrices, so $\op{ad}_X$ is nilpotent for all $X\in\mf g$.
\end{proof}

\subsection{Simple and Semisimple Lie Algebras} \label{subsec:semisimple}
\begin{definition}[simple]
	A Lie algebra $\mf g$ is \textit{simple} if and only if it is not abelian and admits no nonzero proper ideals.
\end{definition}
\begin{example}
	One can check by hand that $\mf{sl}_2(\CC)$ is a simple Lie algebra.
\end{example}
\begin{definition}[semisimple]
	A Lie algebra $\mf g$ is \textit{semismiple} if and only if it admits no nonzero solvable ideals.
\end{definition}
\begin{example}
	Any simple Lie algebra is semisimple. We will have many more examples shortly.
\end{example}
\begin{remark}
	It turns out that any semisimple Lie algebra $\mf g$ is a direct sum of simple ones. In short, one can decompose the adjoint action of $\mf g$ on $\mf g$ into a sum of irreducible subrepresentations, and each irreducible subrepresentation is a simple Lie subalgebra.
\end{remark}
\begin{remark}
	If $\mf g$ is semisimple, then $\mf z(\mf g)=0$, so the adjoint representation $\mf g\to\mf{gl}(\mf g)$ is injective.
\end{remark}
\begin{definition}[reductive]
	A Lie algebra $\mf g$ is \textit{reductive} if and only if $\mf g/\mf z(\mf g)$ is semisimple.
\end{definition}
\begin{example}
	Because $\mf z(\mf{gl}_2(\CC))$ consists of the scalar matrices, the Lie algebra $\mf{gl}_2(\CC)$ is reductive.
\end{example}
\begin{remark}
	A reductive Lie algebra $\mf g$ is semisimple if and only if $\mf z(\mf g)=0$.
\end{remark}
\begin{definition}[radical]
	Fix a Lie algebra $\mf g$. Then the \textit{radical} $\rad\mf g$ is the sum of all solvable ideals of $\mf g$. One can check that $\rad\mf g$ is a solvable ideal.
\end{definition}
\begin{example}
	For a Lie algebra $\mf g$, we see $\rad\mf g=0$ if and only if $\mf g$ is semisimple. For example, $\mf g/\rad\mf g$ is semisimple.
\end{example}
\begin{theorem}[Levi decomposition]
	Fix any Lie algebra $\mf g$. Then there is a semisimple Lie subalgebra $\mf g_{\mathrm{ss}}\subseteq\mf g$ and a direct sum decomposition
	\[\mf g=\rad\mf g\oplus\mf g_{\mathrm{ss}}.\]
\end{theorem}
\begin{proof}[Sketch]
	This is a very difficult result. The idea is to show that the short exact sequence
	\[0\to\rad\mf g\to\mf g\to\mf g_{\mathrm{ss}}\to0\]
	admits a splitting. Such an extension problem reduces to a calculation of some cohomology groups, which is tricky.
\end{proof}
\begin{proposition} \label{prop:get-reductive}
	Let $V$ be a representation of a Lie algebra $\mf g$. Suppose that the pairing $\langle-,-\rangle_V\colon\mf g\times\mf g\to\CC$ defined by
	\[\langle X,Y\rangle_V\coloneqq\tr(XY;V)\]
	is non-degenerate. Then $\mf g$ is reductive.
\end{proposition}
\begin{proof}[Sketch]
	It is enough to show that any $X\in[\mf g,\rad\mf g]$ lives in the kernel of $\langle-,-\rangle_V$ (and thus vanishes). By passing to some short exact sequences, we may assume that $V$ is irreducible. Using \Cref{thm:lie}, one can show that each $\rad(\mf g)$-eigenspace of $V$ is $\mf g$-invariant, so irreducibility implies that $\rad\mf g$ acts by scalars on $V$. A direct calculation now shows that $[\mf g,\rad\mf g]$ acts by zero on $V$.
\end{proof}
\begin{example}
	The standard representation $V=\CC^n$ of $\mf{gl}_n(\CC)$ satisfies the hypothesis of \Cref{prop:get-reductive}, so $\mf{gl}_n(\CC)$ is reductive. A similar argument holds for $\mf{so}_n(\CC)$.
\end{example}
\begin{theorem}[Cartan] \label{thm:cartan-criteria}
	Let $\mf g$ be a semisimple Lie algebra. Then the pairing
	\[\langle X,Y\rangle_{\mf g}\coloneqq\tr(XY;\mf g)\]
	is non-degenerate. This pairing is called the \textit{Killing form}.
\end{theorem}
\begin{proof}[Sketch]
	Please skip this sketch. Let $\mf h$ be the kernel of $\langle-,-\rangle_{\mf g}$, which is an ideal. We will show that $\mf h$ is a solvable, which forces $\mf h=0$. It is enough to show that $\op{ad}(\mf h)\subseteq\mf{gl}(\mf h)$ is solvable. Some linear algebra shows that the pairing $\langle X,Y\rangle\coloneqq\tr(XY;\mf h)$ on $\mf h$ vanishes. Thus, we are in the situation that we have a Lie algebra $\mf h'$ with a faithful representation $V$ for which
	\[\tr(XY;V)=0\]
	for each $X,Y\in\mf h'$. One can use the Jordan decomposition to show that any element of $[\mf h',\mf h']$ is nilpotent, so $[\mf h',\mf h']$ is nilpotent, so $\mf h'$ is solvable.
\end{proof}

\subsection{The Root Decomposition} \label{subsec:root-decomp}
\begin{definition}
	Fix a Lie algebra $\mf g$. Then $X\in\mf g$ is \textit{nilpotent} or \textit{semisimple} if and only if $\op{ad}_X\colon\mf g\to\mf g$ is nilpotent or semisimple, respecively.
\end{definition}
\begin{proposition}[Jordan decomposition]
	Fix a semisimple Lie algebra $\mf g$. For any $X\in\mf g$, there is a unique semisimple $X_s\in\mf g$ and $X_n\in\mf g$ such that $[X_s,X_n]=0$ and $X=X_s+X_n$.
\end{proposition}
\begin{proof}[Sketch]
	The usual Jordan decomposition produces a unique pair $(X_s',X_n')\in\mf{gl}(\mf g)^2$ for which $X_s'$ is semisimple, $X_n'$ is nilpotent, $[X_s',X_n']=0$, and $\mathrm{ad}_X=X_s'+X_n'$. Thus, it remains to check that $X_s'$ and $X_n'$ are in the image of $\op{ad}\colon\mf g\to\mf{gl}(\mf g)$. In short, the Jordan decomposition shows that $X_s'$ and $X_n'$ are polynomials in $\op{ad}_X$, so they are derivations of $\mf g$, but all derivations of $\mf g$ come from $\mf g$.
\end{proof}
\begin{definition}[Cartan]
	Fix a semisimple Lie algebra $\mf g$. Then a \textit{torus} is an abelian subalgebra $\mf h$ of $\mf g$ consisting of semisimple elements. A maximal torus is a \textit{Cartan subalgebra}.
\end{definition}
\begin{example}
	The subalgebra of diagonal matrices of $\mf{sl}_n(\CC)$ is a Cartan subalgebra.
\end{example}
\begin{remark}
	Let $\mf h$ be a Cartan subalgebra. It turns out that
	\[\mf h=\{X\in\mf h:[X,\mf h]=0\}.\]
	In short, use the Jordan decomposition and the maximality of $\mf h$.
\end{remark}
\begin{remark}
	It turns out that any two Cartan subalgebras of $\mf g$ are conjugate under $\op{Aut}\mf g$. This is fairly technical, and we will only use it to show that certain constructions are independent of the choice of $\mf h$.
\end{remark}
\begin{definition}[rank]
	Fix a semisimple Lie algebra $\mf g$. Then the \textit{rank} of $\mf g$ is the dimension of a Cartan subalgebra of $\mf g$.
\end{definition}
\begin{example}
	The rank of $\mf{sl}_n(\CC)$ is $n-1$.
\end{example}
\begin{definition}
	Fix a semisimple Lie algebra $\mf g$ and a Cartan subalgebra $\mf h\subseteq\mf g$. Then the \textit{root system} $\Phi\subseteq\mf h^*$ is the set of nonzero eigenspaces of $\mf h$ acting on $\mf g$.
\end{definition}
\begin{proposition}
	Fix a semisimple Lie algebra $\mf g$ and a Cartan subalgebra $\mf h\subseteq\mf g$. Then the \textit{root decomposition} is the eigenspace decomposition
	\[\mf g=\mf h\oplus\bigoplus_{\alpha\in\Phi}\mf g_\alpha,\]
	where $\mf g_\alpha$ is the eigenspace of $\alpha\in\Phi$.
\end{proposition}
\begin{proof}[Sketch]
	Diagonalize the representation $\mf g$ of $\mf h$.
\end{proof}
\begin{remark}
	For $\alpha,\beta\in\Phi$, we have $[\mf g_\alpha,\mf g_\beta]\subseteq\mf g_{\alpha+\beta}$.
\end{remark}
\begin{remark}
	The subspace $\Phi$ spans $\mf h^*$. Indeed, any $h\in\mf h$ orthogonal to all $\alpha\in\Phi$ then has ${\op{ad}_h}=0$, so $h\in\mf z(\mf g)$, so $h=0$.
\end{remark}
\begin{notation}
	Fix a semisimple Lie algebra $\mf g$ and a Cartan subalgebra $\mf h\subseteq\mf g$ and root system $\Phi\subseteq\mf h^*$. For any $\alpha\in\Phi$, the subspaces $\mf g_\alpha$ and $\mf g_{-\alpha}$ generate a subalgebra of $\mf g$ isomorphic to $\mf{sl}_2(\CC)$. (This is a little tricky to show.) We denote this subalgebra by $\mf{sl}_2(\CC)_\alpha$.
\end{notation}
\begin{proposition}
	Fix a semisimple Lie algebra $\mf g$ and a Cartan subalgebra $\mf h\subseteq\mf g$ and root system $\Phi\subseteq\mf h^*$. Let $(-,-)$ be the Killing form on $\mf g$.
	\begin{listalph}
		\item For each $\alpha\in\Phi$, $\dim\mf g_\alpha=1$.
		\item For each $\alpha,\beta\in\Phi$, $2(\alpha,\beta)/(\alpha,\alpha)\in\ZZ$.
		\item For each $\alpha\in\Phi$, define the reflection $s_\alpha\colon\mf h^*\to\mf h^*$ defined by $s_\alpha(\lambda)\coloneqq\lambda-2(\alpha,\lambda)/(\alpha,\alpha)\cdot\alpha$. Then $s_\alpha$ preserves $\Phi$.
		\item If $\alpha\in\Phi$ and $k\alpha\in\Phi$ for some $k\in\CC$, then $k\in\{+1,-1\}$.
	\end{listalph}
\end{proposition}
\begin{proof}[Sketch]
	Let $(-,-)$ be the Killing form on $\mf h^*$, which is non-degenerate. Define $h_\alpha\coloneqq2H_\alpha/(\alpha,\alpha)$, where $H_\alpha$ is the element dual to $\alpha\in\mf h^*$. Then one can show that
	\[\CC h_\alpha\oplus\bigoplus_{k\in\ZZ\setminus\{0\}}\mf g_{k\alpha}\]
	is an irreducible representation of $\mf{sl}_2(\CC)_\alpha$. Each part now follows from understanding the irredcible representations of $\mf{sl}_2(\CC)$.
\end{proof}
\begin{example}
	Set $\mf g\coloneqq\mf{sl}_n(\CC)$, and let $\mf h$ be the diagonal Cartan subalgebra. For each $i\in\{1,2,\ldots,n\}$, let $e_i\colon\mf h\to\CC$ be the $i$th projection. For each distinct $i$ and $j$, the elementary matrix $E_{ij}$ is an eigenvector for the adjoint $\mf h$-action with eigenvalue $e_i-e_j$. Thus, $\Phi=\{e_i-e_j:i\ne j\}$. Technically, $\mf h^*$ should be thought of as $\CC^n/\CC(e_1+\cdots+e_n)$.
\end{example}
\begin{remark} \label{rem:root-system-to-ss}
	One can recover $\mf g$ from the root system $\Phi\subseteq\mf h^*$. Indeed, $\mf g$ has basis given by a basis of $\mf h^*$ and a vector $e_\alpha$ for each $\alpha\in\Phi$. Roughly speaking, the Lie bracket is determined by requiring $[e_\alpha,e_\beta]\in\CC e_{\alpha+\beta}$ via the $\op{sl}_2(\CC)_\alpha$-representation generated by $e_\beta$. A more precise version of this remark is given by the Serre presentation of a semisimple Lie algebra, which we will not discuss further.
\end{remark}

\subsection{Root Systems} \label{subsec:root-system}
\begin{definition}[reflection]
	A \textit{Euclidean vector space} $E$ is a finite-dimensional real vector space equipped with an inner product $\langle-,-\rangle$. Fix $\alpha\in E$. Then define the \textit{reflection} $s_\alpha\in\op{O}(E)$ as
	\[s_\alpha(\lambda)\coloneqq\lambda-\frac{2(\alpha,\lambda)}{(\alpha,\alpha)}\cdot\alpha\]
\end{definition}
\begin{definition}[root system]
	Fix a Euclidean vector space $E$. A \textit{root system} is a nonempty finite subset $\Phi\subseteq E\setminus\{0\}$ such that
	\begin{listalph}
		\item $\Phi$ spans $E$;
		\item for each $\alpha,\beta\in\Phi$, the number $n_{\alpha\beta}\coloneqq2(\alpha,\beta)/(\beta,\beta)$ is an integer; and
		\item $\Phi$ is stable under $s_\alpha$ for each $\alpha\in\Phi$.
	\end{listalph}
	The root system $\Phi$ is \textit{reduced} if and only if $\ZZ\alpha\cap\Phi=\{\pm\alpha\}$ for each $\alpha\in\Phi$. The \textit{rank} of a root system is $\dim_\RR E$.
\end{definition}
\begin{remark}
	By Cauchy-Schwarz, $n_{\alpha\beta}\in\{-2,-1,-1/2,0,+1/2,+1,+2\}$.
\end{remark}
\begin{example}
	For any semisimple Lie algebra $\mf g$ with Cartan subalgebra $\mf h\subseteq\mf g$. Then the root system $\Phi\subseteq\mf h^*$ is a reduced root system. The root system of $\mf{sl}_n(\CC)$ is denoted $A_{n-1}$.
\end{example}
\begin{example}
	If $\Phi_1\subseteq E_1$ and $\Phi_2\subseteq E_2$ are root systems, then $\Phi_1\sqcup\Phi_2\subseteq E_1\times E_2$ is also a root system. Here, $\Phi_1\subseteq E_1\times E_2$ refers to the subset $\{(\alpha,0)\in E_1\times E_2:\alpha\in\Phi_1\}$.
\end{example}
\begin{definition}[coroot]
	Fix a Euclidean vector space $E$. For $\alpha\in E$, its \textit{coroot} $\alpha^\lor\in E^*$ is defined by
	\[\alpha^\lor(\lambda)\coloneqq\frac{2(\alpha,\lambda)}{(\alpha,\alpha)}.\]
\end{definition}
\begin{definition}
	Fix root systems $\Phi_1\subseteq E_1$ and $\Phi_2\subseteq E_2$. An \textit{isomorphism} of root systems is a bijection $\varphi\colon\Phi_1\to\Phi_2$ which extends to a linear map $E_1\to E_2$ such that
	\[\frac{2(\alpha,\beta)}{(\beta,\beta)}=\frac{2(\varphi(\alpha),\varphi(\beta))}{(\varphi(\beta),\varphi(\beta))}.\]
\end{definition}
\begin{remark}
	The linear map $\varphi\colon E_1\to E_2$ does not have to preserve the inner product. For example, one could simply define $E_2$ to have twice the inner product of $E_1$ (and the same underlying vector space).
\end{remark}
\begin{example}
	Here are all the root systems of rank two, up to isomorphism, drawn in $\RR^2$.
\begin{center}
\begin{asy}
	unitsize(1.5cm);

\begin{definition}[irreducible]
	A root system $\Phi$ is \textit{irreducible} if and only if it is not isomorphic to a root system of the form $\Phi_1\sqcup\Phi_2$, where $\Phi_1$ and $\Phi_2$ are root systems.
\end{definition}
\begin{remark}
	It turns out that any root system is the disjoint union of irreducible root systems.
\end{remark}
\begin{definition}[Weyl group]
	Fix a root system $\Phi\subseteq E$. Then the \textit{Weyl group} $W$ of $\Phi$ is the subgroup of $\op{GL}(E)$ generated by $\{s_\alpha:\alpha\in\Phi\}$. The \textit{sign} of $w\in W$, denoted $\det w$, is equal to the determinant of its action on $E$.
\end{definition}
\begin{example}
	The Weyl group of $A_n$ is $S_{n-1}$.
\end{example}

\subsection{Dynkin Diagrams} \label{subsec:dynkin}
\begin{definition}[polarization]
	Fix a root system $\Phi\subseteq E$. A \textit{polarization} of $\Phi$ is a choice of $t\in E$ such that $(t,\alpha)\ne0$ for all $\alpha\in\Phi$. The pair $(\Phi,t)$ is a \textit{polarized root system}, though we will frequently omit the choice of $t$.
\end{definition}
\begin{definition}[positive]
	Fix a polarized root system $\Phi\subseteq E$. Then a root $\alpha\in\Phi$ is \textit{positive} if and only if $(t,\alpha)>0$. The set of positive roots is denoted $\Phi^+$. One similarly defines the set of negative roots $\Phi^-$.
\end{definition}
\begin{example}
	The element $t\coloneqq(1,2,\ldots,n)$ is a polarization of $A_n$. Then the positive roots are $\{e_i-e_j:i<j\}$.
\end{example}
\begin{definition}[simple]
	Fix a polarized root system $\Phi\subseteq E$. Then a positive root $\alpha\in\Phi$ is \textit{simple} if and only if it is not the sum of more than one positive root.
\end{definition}
\begin{remark}
	It turns out that the set of simple roots is a basis of $E$. Also, any positive root can be written as a sum of simple roots with nonnegative integral coefficients.
\end{remark}
\begin{example}
	The simple roots of $A_n$ are $\{e_i-e_{i+1}:1\le i<n\}$.
\end{example}
\begin{lemma}
	Fix a root system $\Phi\subseteq E$. Then one can recover $\Phi$ from the subset $\Delta\subseteq\Phi$ of simple roots.
\end{lemma}
\begin{proof}[Sketch]
	Let $W$ be the Weyl group of $\Phi$. It turns out that the set $\{s_\alpha:\alpha\in\Delta\}$ of ``simple reflections'' generates $W$, and $W(\Delta)=\Phi$.
\end{proof}
\begin{definition}[Dynkin diagram]
	Fix a polarized root system $\Phi\subseteq E$, and enumerate the simple roots as $\{\alpha_1,\ldots,\alpha_n\}$. Then the \textit{Dynkin diagram} of $\Phi$ is the directed multigraph with the following data.
	\begin{itemize}
		\item The vertices of $\Phi$ is the index set $\{1,\ldots,n\}$.
		\item Vertices $i$ and $j$ are connected by $a_{ij}a_{ji}$, with the edge directed from $i$ to $j$ if and only if $(\alpha_i,\alpha_i)>(\alpha_j,\alpha_j)$.
	\end{itemize}
\end{definition}
\begin{remark}
	The Dynkin diagram $D$ of a root system $\Phi$ is disconnected if and only if $\Phi$ is reducible, and each connected component of $D$ corresponds to an irreducible component of $\Phi$.
\end{remark}
\begin{example}
	The Dynkin diagram of $A_n$ is the path with $n$ vertices.
\end{example}
\begin{lemma} \label{lem:root-to-dynkin}
	Fix a root system $\Phi\subseteq E$. Then one can recover $\Phi$ from its Dynkin diagram.
\end{lemma}
\begin{proof}[Sketch]
	It is enough to recover the simple roots. Well, the Dynkin diagram remembers the number of simple roots and the numbers $n_{ij}\coloneqq2(\alpha_i,\alpha_j)/(\alpha_j,\alpha_j)$, which is enough to recover the roor system up to isomorphism.
\end{proof}

\subsection{Highest Weight Theory} \label{subsec:highest-weight}
\begin{conv} \label{conv:ss-rep-theory}
	Throughout this section, $\mf g$ is a semisimple Lie algebra with a Cartan subalgebra $\mf h\subseteq\mf g$. The polarized root system is $\Phi\subseteq\mf h^*$, and its Weyl group is $W$. Representations need not be finite-dimensional.
\end{conv}
\begin{theorem}
	All finite-dimensional representations of $\mf g$ are completely reducible.
\end{theorem}
\begin{proof}[Sketch]
	There are many proofs of this theorem. The main idea is to show that any short exact sequence
	\[0\to V'\to V\to V''\to0\]
	of representations of $\mf g$ splits as a direct sum $V=V'\oplus V''$. The most algebraic proof proceeds by finding ``Casimir'' elements in the center of $U\mf g$. By \Cref{lem:schur}, they act by scalars on irreducible representations. One can force the splitting if granted a Casimir elements which acts on $V'$ and $V''$ by different scalars. This turns out to be possible as long as $V'$ and $V''$ are not isomorphic. The case where $V'$ and $V''$ are isomorphic is handled directly.
\end{proof}
\begin{definition}[weight decomposition]
	A \textit{weight} is an element of $\mf h^*$. For a representation $V$ of $\mf g$, a \textit{weight vector} is an eigenvector for $\mf h$, and a \textit{weight decomposition} is an eigenspace decomposition of $\mf h$ acting on $V$. For $\lambda\in\mf h^*$, we define $V[\lambda]$ to be the $\lambda$-eigenspace, meaning
	\[V[\lambda]\coloneqq\{v\in V:Xv=\lambda(X)v\text{ for all }X\in\mf h\}.\]
	The multiset of $\lambda\in\mf h^*$ for which $V[\lambda]\ne0$ are referred to as the \textit{weights} of $V$, where multiplicities are by $\dim_\CC V[\lambda]$.
\end{definition}
\begin{definition}[integral]
	A weight $\lambda\in\mf h^*$ is \textit{integral} if and only if $\alpha^\lor(\lambda)\in\ZZ$ for all . The subgroup of integral weights is denoted $P\subseteq\mf h^*$. While we're here, we define the \textit{root lattice} $Q\subseteq\mf h^*$.
\end{definition}
\begin{remark}
	The definition of the root system implies that $Q\subseteq P$. The quotient $P/Q$ is interesting.
\end{remark}
\begin{example}
	For $\mf g=\mf{sl}_n(\CC)$, one has $P=\{(\lambda_1,\ldots,\lambda_n):\lambda_i-\lambda_j\in\ZZ\text{ for all }i,j\}$ and $Q=\{(\lambda_1,\ldots,\lambda_n):\sum_i\lambda_i=0\}$. Thus, $P/Q\cong\ZZ/n\ZZ$.
\end{example}
\begin{lemma}
	Any finite-dimensional representation $V$ of $\mf g$ admits a weight decomposition by integral weights. The multiset of weights of $V$ is $W$-invariant.
\end{lemma}
\begin{proof}[Sketch]
	Let's start with the weight decomposition. Because $\mf h$ is abelian and spanned by $\{h_\alpha\}_{\alpha\in\Phi}$, it is enough to show that $h_\alpha$ acts semisimply on $V$ for each $\alpha\in\Phi$. For a given $\alpha\in\Phi$, this follows by the representation theory of $\mf{sl}_2(\CC)$ upon restricting to $\mf{sl}_2(\CC)_\alpha\subseteq\mf g$. Integrality follows because $h_\alpha$ acts on $V[\lambda]$ by $\alpha^\lor(\lambda)$, which must be an integer. To check invariance under $W$, it is enough to check invariance under each reflection $s_\alpha$ for $\alpha\in\Phi$, which follows by the symmetry of the weights of the irreducible representations of $\mf{sl}_2(\CC)_\alpha$.
\end{proof}
\begin{notation}
	Define $\mf n_+$ to be
	\[\bigoplus_{\alpha\in\Phi^+}\mf g_\alpha\subseteq\mf g.\]
	Define $\mf n_-$ similarly. We may abbreivate $\mf n_+$ to $\mf n$. These are Lie subalgebras of $\mf g$.
\end{notation}
\begin{definition}[singular]
	Fix a representation $V$ of $\mf g$. A vector $v\in V$ is \textit{singular} if and only if $v$ is a nonzero weight vector and $\mf n\cdot v=0$. The representation $V$ is a \textit{highest weight} representation of weight $\lambda$ if and only if $V$ is generated by a singular vector of weight $\lambda$.
\end{definition}
\begin{lemma}
	Fix an irreducible finite-dimensional representation $V$ of $\mf g$. Then $V$ is a highest weight representation.
\end{lemma}
\begin{proof}[Sketch]
	By irreducibility, $V$ is generated by any nonzero vector, so it is enough to exhibit a nonzero singular vector. Select the weight $\lambda$ in $V$ which is ``maximal'' in the sense that $\alpha+\lambda$ is not a weight in $V$ for any $\alpha\in\Phi^+$. Then any nonzero $v\in V[\lambda]$ will do.
\end{proof}
\begin{definition}[Verma module]
	Fix a weight $\lambda\in\mf h^*$. Then define the \textit{Verma module} $M_\lambda$ to be $U\mf g\otimes_{\mf b}\CC_\lambda$, where $\mf b=\mf h\oplus\mf n$. For brevity, we define $v_\lambda\in M_\lambda$ to be the vector $1\otimes1$.
\end{definition}
\begin{remark}
	By \Cref{thm:pbw}, the natural map $U\mf n_-\to U\mf g$ induces an isomorphism $U\mf n_-\to M_\lambda$ of vector spaces, meaning that $M_\lambda$ admits a basis by ordered monomials in $\mf n_-$.
\end{remark}
\begin{lemma}
	Fix a representation $V$ of highest weight $\lambda$. For any singular vector $v\in V[\lambda]$, there is a unique surjection $M_\lambda\to V$ such that $v_\lambda\mapsto v$.
\end{lemma}
\begin{proof}[Sketch]
	There is a natural map $U\mf g\to V$ sending $1\mapsto v$. Because $v$ is a highest weight vector of weight $\lambda$, this natural map is surjective and descends to $M_\lambda$.
\end{proof}
\begin{lemma}
	Fix $\lambda\in\mf h^*$. Then $M_\lambda$ admits a unique maximal proper submodule.
\end{lemma}
\begin{proof}[Sketch]
	A submodule $N$ of $M_\lambda$ is proper if and only if $v_\lambda\notin N$, so the maximal submodule is simply the sum of all submodules not containing $v_\lambda$.
\end{proof}
\begin{notation}
	Fix a weight $\lambda\in\mf h^*$. Then define $L_\lambda$ to the quotient of $M_\lambda$ by its maximal proper submodule.
\end{notation}
\begin{remark}
	Assembling our results shows that any finite-dimensional irreducible representation $V$ is of highest weight $\lambda$ for some weight $\lambda$ and thus must be isomorphic to $L_\lambda$.
\end{remark}
\begin{definition}[dominant]
	A weight $\lambda\in\mf h^*$ is \textit{dominant} if and only if $\alpha^\lor(\lambda)\ge0$ for each $\alpha\in\Phi^+$.
\end{definition}
\begin{theorem}[Highest weight]
	For $\lambda\in\mf h^*$, the representation $L_\lambda$ is finite-dimensional if and only if $\lambda$ is dominant and integral. Thus, finite-dimensional irreducible representations are in bijection with dominant integral weights.
\end{theorem}
\begin{proof}[Sketch]
	If $L_\lambda$ is finite-dimensional, then for each $\alpha\in\Phi^+$, one knows that $v_\lambda$ generates a finite-dimensional representation of $\mf{sl}_2(\CC)_\alpha$ with $e_\alpha v_\lambda=0$ and $h_\alpha v_\lambda=\alpha^\lor(\lambda)v_\lambda$. Comparing this with the known representations of $\mf{sl}_2(\CC)$ forces $\alpha^\lor(\lambda)$ to be a nonnegative integer.
	
	Conversely, suppose $\lambda$ is dominant and integral, so we must check $\dim_\CC L_\lambda<\infty$. This is long and difficult, so we will say very little. In short, for each simple reflection $s$, one can show that $M_{s\lambda}$ embeds into $M_\lambda$, and then one should check that the quotient of $M$ by the sum of the $M_{s\lambda}$s is finite-dimensional.
\end{proof}
\begin{remark}
	Though we have classified the finite-dimensional irreducible representations, this construction is highly non-explicit. It is possible but difficult to even get rough information about these representations; see \Cref{thm:weyl-character} and \Cref{cor:weyl-dim}.
\end{remark}

\subsection{Character Theory} \label{subsec:char}
\begin{conv}
	\Cref{conv:ss-rep-theory} continues to apply.
\end{conv}
\begin{definition}[character]
	Fix a representation $V$ of $\mf g$. Then the \textit{character} of $V$ is the formal power series
	\[\op{ch}V\coloneqq\sum_{\lambda\in\mf h^*}\dim_\CC V[\lambda]\cdot e^\lambda.\]
	Here, $e^\lambda$ is a formal symbol satisfying the usual exponent rule $e^{\lambda+\mu}=e^\lambda e^\mu$ for $\lambda,\mu\in\mf h^*$.
\end{definition}
\begin{example}
	Let $\CC$ be the trivial representation. Then $\op{ch}\CC=1$.
\end{example}
\begin{example}
	Let $\CC^2$ be the standard representation of $\mf{sl}_2(\CC)$. Then $\op{ch}\CC^2=e^{-\alpha/2}+e^{\alpha/2}$, where $\alpha\in\mf h$ is the unique positive root. More generally,
	\[\op{ch}\op{Sym}\CC^2=e^{-n\alpha/2}+e^{-n\alpha/2+1}+\cdots+e^{n\alpha/2-1}+e^{n\alpha/2}.\]
\end{example}
\begin{example}
	For any representation $V$, one has $\op{ch}V^\lor=\overline{\op{ch}V}$, where the overline means $\overline{e^\lambda}=e^{-\lambda}$.
\end{example}
\begin{example}
	For representations $V$ and $W$, one has $\op{ch}(V\oplus W)=\op{ch}V+\op{ch}W$. More generally, given a short exact sequence
	\[0\to V'\to V\to V''\to0\]
	of representations admitting weight decompositions, one has $\op{ch}V=\op{ch}V'+\op{ch}V''$.
\end{example}
\begin{example}
	For representations $V$ and $W$, one has $\op{ch}(V\otimes W)=\op{ch}V\cdot\op{ch}W$.
\end{example}
\begin{example} \label{ex:ch-verma}
	Fix $\lambda\in\mf h^*$. By expanding out the basis of $M_\lambda$, one finds that
	\[\op{ch}M_\lambda=e^\lambda\prod_{\alpha\in\Phi^+}\frac1{1-e^{-\alpha}}.\]
\end{example}
\begin{notation}
	Define $\rho\coloneqq\frac12\sum_{\alpha\in\Phi^+}\alpha$.
\end{notation}
\begin{theorem}[Weyl character formula] \label{thm:weyl-character}
	Fix a dominant integral weight $\lambda$. Then
	\[\op{ch}L_\lambda=\sum_{w\in W}\det(w)e^{w(\lambda+\rho)}\mathbin{\bigg/}\prod_{\alpha\in\Phi^+}\big(e^{\alpha/2}-e^{-\alpha/2}\big).\]
\end{theorem}
\begin{proof}[Sketch]
	This is another long and difficult theorem. One approach is to build an exact sequence of the form
	\[0\to{?}\to{?}\to\cdots\to{?}\to L_\lambda\to0,\]
	where each $?$ is a sum of Verma modules. For example, there is a natural surjection $M_\lambda\to L_\lambda$, and the kernel admits a natural surjection of the form $\bigoplus_i M_{s_i\lambda}$. Once one fills in each $?$, one can apply \Cref{ex:ch-verma}.
\end{proof}
\begin{example}[Weyl demoninator formula] \label{ex:weyl-denom}
	When $\lambda=0$, we find $L_0=\CC$, so $\op{ch}L_0=1$. Thus,
	\[\sum_{w\in W}\det(w)e^{w\rho}=\prod_{\alpha\in\Phi^+}\big(e^{\alpha/2}-e^{-\alpha/2}\big).\]
\end{example}
\begin{corollary}[Weyl dimension formula] \label{cor:weyl-dim}
	Fix a dominant integral weight $\lambda$. Then
	\[\dim L_\lambda=\prod_{\alpha\in\Phi^+}\frac{(\lambda+\rho,\alpha)}{(\rho,\alpha)}.\]
\end{corollary}
\begin{proof}[Sketch]
	Fix a formal variable $q$, and we will compute
	\[\dim_qL_\lambda\coloneqq\sum_{\mu\in\mf h^*}\dim_\CC L_\lambda[\mu]\cdot q^{2(\rho,\lambda)},\]
	from which the dimension can be computed by sending $q\to1$. There is a homomorphism $\CC[e^P]\to\CC[q,1/q]$ given by sending $e^\lambda$ to $q^{2(\lambda,\rho)}$, and it sends $\op{ch}L_\lambda$ to $\dim_qL_\lambda$. One now computes $\dim_qL_\lambda$ by using \Cref{thm:weyl-character}: note
	\[\dim_qL_\lambda=\sum_{w\in W}\det(w)q^{(w(\lambda+\rho),\rho)}\mathbin{\bigg/}\prod_{\alpha\in\Phi^+}\big(q^{(\rho,\alpha)}-q^{-(\rho,\alpha)}\big).\]
	The numerator can be simplified with \Cref{ex:weyl-denom}, and then we may take $q\to1$.
\end{proof}
